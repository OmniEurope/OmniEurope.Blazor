Sécurité : résultats et limites

Aucune faille de sécurité confirmée dans les chemins examinés. Le contrôle NuGet en ligne ne retourne aucune vulnérabilité connue pour les dépendances directes ou transitives des 12 projets de la solution. Cette observation est datée du 11 octobre 2026 et ne couvre pas les dépendances du projet Hybrid hors solution ni les dépendances futures.

Chemins relus : garde des attributs CSP, vérification des URI, liens externes, returnUrl local, nettoyage HTML et règles des extensions, collage de l’éditeur, encodage HTML des tableaux importés, protection des formules CSV et écriture des cellules XLSX, lecture bornée des fichiers, persistance navigateur et workflows de livraison. Le scan CSP et ses fixtures passent ; les tests de ces chemins sont inclus dans la suite exécutée.

Le réemploi d’une instance HtmlSanitizer configurée une fois n’est pas en soi un défaut de concurrence, selon la documentation de son auteur. Les métadonnées IBrowserFile restent des entrées non fiables : l’hôte doit vérifier contenu, droits et quotas avant stockage, conformément à la documentation Microsoft. Le filtrage MIME dans OmniUpload est un contrôle d’interface, pas une authentification du contenu.

Non audités ici : infrastructure en production, authentification et autorisations des applications consommatrices, secrets externes, analyse dynamique complète XSS dans plusieurs navigateurs, résistance à toutes les entrées hostiles de toutes les extensions. La CSP dépend des en-têtes de l’hôte ; l’exception Monaco est explicitement documentée.

using Bunit;
using Microsoft.AspNetCore.Components.Forms;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.JSInterop;
using OmniEurope.Blazor;
using OmniEurope.Blazor.Components;

using (var context = CreateContext())
{
    var pending = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
    OmniUploadRequest? requestSeen = null;
    IReadOnlyList<OmniUploadFile> attached = [];
    var upload = context.Render<OmniUpload>(parameters => parameters
        .Add(component => component.FilesChanged, files => attached = files)
        .Add(component => component.Upload, request => { requestSeen = request; return pending.Task; }));
    var input = upload.FindComponent<InputFile>();
    var selecting = Task.Run(() => input.UploadFiles(InputFileContent.CreateFromText("hello", "a.txt", contentType: "text/plain")));
    upload.WaitForAssertion(() => Require(upload.FindAll(".omni-upload__cancel").Count == 1, "No cancel button"));
    upload.Find(".omni-upload__cancel").Click();
    Require(requestSeen!.CancellationToken.IsCancellationRequested, "Token not cancelled");
    pending.SetResult();
    await selecting.WaitAsync(TimeSpan.FromSeconds(10));
    var message = upload.Find(".omni-upload__message").TextContent;
    Require(attached.Count == 1 && message.Contains("terminé", StringComparison.OrdinalIgnoreCase), "Unexpected cancellation outcome: " + message);
    Console.WriteLine($"UPLOAD_CANCEL: tokenCancelled=True, attached={attached.Count}, message={message}");
}

using (var context = CreateContext())
{
    context.JSInterop.Setup<string?>("localStorage.getItem", _ => true).SetException(new JSException("SecurityError: storage denied"));
    try
    {

        throw new InvalidOperationException("Storage error was unexpectedly contained");
    }
    catch (JSException exception)
    {
        Console.WriteLine("GRID_STORAGE: escaped=" + exception.Message);
    }
}

using (var overlays = new OmniOverlayService())
{
    overlays.Notify("short", duration: TimeSpan.FromMilliseconds(30));
    using var iterator = overlays.Notifications.GetEnumerator();
    Require(iterator.MoveNext(), "Missing notification");
    await Task.Delay(100);
    try
    {
        iterator.MoveNext();
        throw new InvalidOperationException("Notification collection was unexpectedly stable");
    }
    catch (InvalidOperationException exception) when (exception.Message.Contains("Collection was modified", StringComparison.Ordinal))
    {
        Console.WriteLine("NOTIFICATION_ENUMERATION: " + exception.Message);
    }
}

static BunitContext CreateContext()
{
    var context = new BunitContext();
    context.JSInterop.Mode = JSRuntimeMode.Loose;
    context.Services.AddLogging();
    context.Services.AddOmniEuropeBlazor();
    return context;
}

static void Require(bool condition, string message)
{
    if (!condition) throw new InvalidOperationException(message);
}
